```python
"""Observation builders for all 5 obs_modes.

Called by UAVPlacementEnv._obs(). Returns ndarray for flat/sorted/cluster_summary,
dict[str, ndarray] for spatial_grid and graph.
"""
from __future__ import annotations

import numpy as np
from gymnasium import spaces


MAX_OBS_USERS = 150  # hard ceiling — matches max n_users in RQ3-A experiments


# ── vector obs modes ─────────────────────────────────────────────────────────

def build_flat(uav_norm: np.ndarray, users: np.ndarray, cfg) -> np.ndarray:
    return np.concatenate([uav_norm.ravel(), (users / cfg.area).ravel()]).astype(np.float32)


def build_sorted(uav_norm: np.ndarray, users: np.ndarray, cfg) -> np.ndarray:
    centroid = np.full(2, cfg.area / 2.0)
    order = np.argsort(np.linalg.norm(users - centroid, axis=1))
    return np.concatenate([uav_norm.ravel(), (users[order] / cfg.area).ravel()]).astype(np.float32)


def build_cluster_summary(uav_norm: np.ndarray, users: np.ndarray, cfg) -> np.ndarray:
    from sklearn.cluster import KMeans
    km = KMeans(n_clusters=cfg.n_clusters, n_init=1, random_state=0)
    labels = km.fit_predict(users)
    counts = np.bincount(labels, minlength=cfg.n_clusters)
    summary = np.column_stack([
        km.cluster_centers_ / cfg.area,
        counts.astype(float) / cfg.n_users,
    ])
    return np.concatenate([uav_norm.ravel(), summary.ravel()]).astype(np.float32)


# ── spatial_grid ─────────────────────────────────────────────────────────────

def build_spatial_grid(uavs: np.ndarray, users: np.ndarray, cfg) -> dict:
    """32×32 binary occupancy grid of users + normalised UAV vector in [-1, 1]."""
    grid = np.zeros((1, 32, 32), dtype=np.float32)
    idx = np.floor(users / cfg.area * 32).astype(int).clip(0, 31)
    grid[0, idx[:, 1], idx[:, 0]] = 1.0  # row=y, col=x

    u = uavs.astype(np.float64).copy()
    u[:, :2] = u[:, :2] / cfg.area * 2.0 - 1.0
    u[:, 2] = (u[:, 2] - cfg.h_min) / max(cfg.h_max - cfg.h_min, 1e-6) * 2.0 - 1.0
    return {"grid": grid, "uavs": u.ravel().astype(np.float32)}


# ── graph ─────────────────────────────────────────────────────────────────────

def build_graph(
    uavs: np.ndarray,
    users: np.ndarray,
    cfg,
    failed_mask: np.ndarray,
) -> dict:
    """Fixed-capacity graph tensors for PyG/SB3.

    Layout: UAV nodes at indices 0..max_fleet_size-1,
            user nodes at indices max_fleet_size..max_fleet_size+MAX_OBS_USERS-1,
            padding zeros beyond.
    """
    max_fleet = cfg.max_fleet_size
    max_nodes = max_fleet + MAX_OBS_USERS
    max_edges = 4000

    n_uavs = uavs.shape[0]
    n_users = users.shape[0]

    # node features: [x/area, y/area, z/h_max, is_uav]
    node_feats = np.zeros((max_nodes, 4), dtype=np.float32)
    for i in range(n_uavs):
        node_feats[i] = [uavs[i, 0] / cfg.area, uavs[i, 1] / cfg.area,
                         uavs[i, 2] / cfg.h_max, 1.0]
    for j in range(n_users):
        node_feats[max_fleet + j] = [users[j, 0] / cfg.area, users[j, 1] / cfg.area, 0.0, 0.0]

    # node_type: 0=padding, 1=UAV, 2=user
    node_type = np.zeros(max_nodes, dtype=np.float32)
    node_type[:n_uavs] = 1.0
    node_type[max_fleet : max_fleet + n_users] = 2.0

    # uav_active: 1=active, 0=failed or padding
    uav_active = np.zeros(max_fleet, dtype=np.float32)
    for i in range(n_uavs):
        uav_active[i] = float(failed_mask[i])

    # edges: UAV-UAV fully connected + UAV-User within graph_edge_radius
    src, dst = [], []
    for i in range(n_uavs):
        for j in range(n_uavs):
            if i != j:
                src.append(i); dst.append(j)
    radius = cfg.graph_edge_radius
    for i in range(n_uavs):
        dists = np.linalg.norm(users - uavs[i, :2], axis=1)
        for j in np.where(dists <= radius)[0]:
            u_node = max_fleet + j
            src += [i, u_node]; dst += [u_node, i]

    n_edges = min(len(src), max_edges)
    edge_index = np.zeros((2, max_edges), dtype=np.float32)
    if n_edges > 0:
        edge_index[0, :n_edges] = np.array(src[:n_edges], dtype=np.float32)
        edge_index[1, :n_edges] = np.array(dst[:n_edges], dtype=np.float32)

    return {
        "node_feats": node_feats,
        "edge_index": edge_index,
        "node_type": node_type,
        "uav_active": uav_active,
        "n_nodes": np.array([float(n_uavs + n_users)], dtype=np.float32),
        "n_edges": np.array([float(n_edges)], dtype=np.float32),
    }


# ── observation space factory ─────────────────────────────────────────────────

def make_obs_space(cfg) -> spaces.Space:
    U, K = cfg.n_uavs, cfg.n_users
    if cfg.obs_mode == "flat":
        return spaces.Box(0.0, 1.0, shape=(U * 3 + K * 2,), dtype=np.float32)
    if cfg.obs_mode == "sorted":
        return spaces.Box(0.0, 1.0, shape=(U * 3 + K * 2,), dtype=np.float32)
    if cfg.obs_mode == "cluster_summary":
        return spaces.Box(0.0, 1.0, shape=(U * 3 + cfg.n_clusters * 3,), dtype=np.float32)
    if cfg.obs_mode == "spatial_grid":
        return spaces.Dict({
            "grid": spaces.Box(0.0, 1.0, shape=(1, 32, 32), dtype=np.float32),
            "uavs": spaces.Box(-1.0, 1.0, shape=(U * 3,), dtype=np.float32),
        })
    if cfg.obs_mode == "graph":
        mf = cfg.max_fleet_size
        mn = mf + MAX_OBS_USERS
        return spaces.Dict({
            "node_feats": spaces.Box(-1.0, 1.0,          shape=(mn, 4),    dtype=np.float32),
            "edge_index": spaces.Box(0.0,  float(mn),    shape=(2, 4000),  dtype=np.float32),
            "node_type":  spaces.Box(0.0,  2.0,          shape=(mn,),      dtype=np.float32),
            "uav_active": spaces.Box(0.0,  1.0,          shape=(mf,),      dtype=np.float32),
            "n_nodes":    spaces.Box(0.0,  float(mn),    shape=(1,),       dtype=np.float32),
            "n_edges":    spaces.Box(0.0,  4000.0,       shape=(1,),       dtype=np.float32),
        })
    raise ValueError(f"Unknown obs_mode: {cfg.obs_mode!r}")
```

```python
"""CNN feature extractor for spatial_grid obs_mode.

Input Dict: {"grid": (1, 32, 32), "uavs": (3*n_uavs,)}
Output: (batch, features_dim)  default features_dim=128

Compatible with SB3 MultiInputPolicy via policy_kwargs:
    {"features_extractor_class": CnnExtractor,
     "features_extractor_kwargs": {"features_dim": 128}}
"""
from __future__ import annotations

import gymnasium as gym
import torch
import torch.nn as nn
from stable_baselines3.common.torch_layers import BaseFeaturesExtractor


class CnnExtractor(BaseFeaturesExtractor):
    def __init__(self, observation_space: gym.spaces.Dict, features_dim: int = 128):
        super().__init__(observation_space, features_dim)
        uav_dim = observation_space["uavs"].shape[0]
        self.cnn = nn.Sequential(
            nn.Conv2d(1, 16, kernel_size=3, padding=1), nn.ReLU(),
            nn.Conv2d(16, 32, kernel_size=3, padding=1), nn.ReLU(),
            nn.AdaptiveAvgPool2d((4, 4)),
            nn.Flatten(),           # → 32×4×4 = 512
        )
        self.fc = nn.Linear(512 + uav_dim, features_dim)

    def forward(self, observations: dict) -> torch.Tensor:
        grid_feat = self.cnn(observations["grid"])          # (batch, 512)
        uav_feat = observations["uavs"].float()             # (batch, 3*n_uavs)
        return torch.relu(self.fc(torch.cat([grid_feat, uav_feat], dim=-1)))
```

```python
"""GATv2-based feature extractor for graph obs_mode.

Architecture:
  GATv2Conv(4 → 64, heads=4, concat=True)   → (n_actual_nodes, 256)
  GATv2Conv(256 → 64, heads=1, concat=False) → (n_actual_nodes, 64)
  Extract UAV nodes; apply uav_active mask → zero out failed/padding slots
  Flatten: (max_fleet * 64,) per sample

Each sample processed separately (PyG variable-size graphs vs SB3 fixed batches).
Acceptable overhead for research use at batch_size ≤ 256.

Requires: torch_geometric>=2.5  (pip install torch_geometric)
"""
from __future__ import annotations

import gymnasium as gym
import torch
import torch.nn.functional as F
from stable_baselines3.common.torch_layers import BaseFeaturesExtractor
from torch_geometric.nn import GATv2Conv


class GNNExtractor(BaseFeaturesExtractor):
    UAV_EMB_DIM = 64

    def __init__(self, observation_space: gym.spaces.Dict, features_dim: int | None = None):
        self.max_fleet = observation_space["uav_active"].shape[0]
        inferred = self.max_fleet * self.UAV_EMB_DIM
        if features_dim is None:
            features_dim = inferred
        if features_dim != inferred:
            raise ValueError(f"features_dim must equal max_fleet*64={inferred}, got {features_dim}")
        super().__init__(observation_space, features_dim)
        self.conv1 = GATv2Conv(in_channels=4,   out_channels=64, heads=4, concat=True)   # → 256
        self.conv2 = GATv2Conv(in_channels=256, out_channels=64, heads=1, concat=False)  # → 64

    def forward(self, observations: dict) -> torch.Tensor:
        batch_size = observations["node_feats"].shape[0]
        device = observations["node_feats"].device
        mf, ed = self.max_fleet, self.UAV_EMB_DIM
        results = []
        for b in range(batch_size):
            n = int(observations["n_nodes"][b, 0].item())
            ne = int(observations["n_edges"][b, 0].item())
            x = observations["node_feats"][b, :n].float()
            if ne > 0:
                ei = observations["edge_index"][b, :, :ne].long()
            else:
                ei = torch.zeros(2, 0, dtype=torch.long, device=device)
            x = F.elu(self.conv1(x, ei))    # (n, 256)
            x = self.conv2(x, ei)           # (n, 64)
            uav_emb = torch.zeros(mf, ed, device=device)
            n_uavs = min(mf, n)
            uav_emb[:n_uavs] = x[:n_uavs]
            uav_active = observations["uav_active"][b].float()
            uav_emb = uav_emb * uav_active.unsqueeze(-1)
            results.append(uav_emb.flatten())
        return torch.stack(results, dim=0)   # (batch, max_fleet*64)
```

```python
def warm_start_pso_planner(
    env: UAVPlacementEnv,
    rng: np.random.Generator,
    prev_solution: np.ndarray | None = None,
    n_particles: int = 30,
    iters: int = 60,
    w: float = 0.7,
    c1: float = 1.5,
    c2: float = 1.5,
) -> np.ndarray:
    """PSO warm-started from a prior solution.

    One particle is seeded with prev_solution (flattened); the rest are random.
    Same computational budget as cold-start PSO.
    """
    c = env.cfg
    lo = np.tile([0.0, 0.0, c.h_min], c.n_uavs)
    hi = np.tile([c.area, c.area, c.h_max], c.n_uavs)
    dim = lo.size
    x = rng.uniform(lo, hi, size=(n_particles, dim))
    if prev_solution is not None:
        x[0] = np.clip(prev_solution.ravel(), lo, hi)
    v = rng.normal(0.0, 0.1, size=(n_particles, dim)) * (hi - lo)
    score = lambda p: _soft_coverage(env, p.reshape(c.n_uavs, 3))
    pbest = x.copy()
    pbest_f = np.array([score(p) for p in x])
    g = pbest[np.argmax(pbest_f)].copy()
    for _ in range(iters):
        r1, r2 = rng.random((2, n_particles, dim))
        v = w * v + c1 * r1 * (pbest - x) + c2 * r2 * (g - x)
        x = np.clip(x + v, lo, hi)
        f = np.array([score(p) for p in x])
        better = f > pbest_f
        pbest[better], pbest_f[better] = x[better], f[better]
        g = pbest[np.argmax(pbest_f)].copy()
    return g.reshape(c.n_uavs, 3)


def mpc_planner(
    env: UAVPlacementEnv,
    rng: np.random.Generator,
    horizon: int = 5,
    budget_ms: float = 50.0,
    pso_particles: int = 10,
) -> np.ndarray:
    """Receding-horizon MPC: time-budgeted PSO re-plan.

    Runs PSO with pso_particles scaled to fit within budget_ms wall-clock.
    Calibrates iters on a trial call; returns targets for use with TargetTrackingPolicy.
    """
    import time
    c = env.cfg
    t0 = time.perf_counter()
    pso_planner(env, rng, n_particles=pso_particles, iters=horizon)
    elapsed_ms = (time.perf_counter() - t0) * 1000.0
    iters = max(horizon, int(horizon * budget_ms / max(elapsed_ms, 0.1)))
    return pso_planner(env, rng, n_particles=pso_particles, iters=iters)
```

```python
"""Statistical helpers for RQ1-RQ3 significance testing.

Primary test: paired difference CI — NOT single-metric CI overlap.

Usage:
    from scripts.stats import paired_diff_ci, prob_of_improvement

    point, lo, hi = paired_diff_ci(rl_scores, pso_scores)
    # Significant RL improvement if lo > 0
    # Significant PSO improvement if hi < 0

    poi = prob_of_improvement(rl_scores, pso_scores)
    # 1.0 means RL > PSO on every sample; 0.5 means indistinguishable
"""
from __future__ import annotations

import numpy as np


def paired_diff_ci(
    scores_a: np.ndarray,
    scores_b: np.ndarray,
    reps: int = 5_000,
    ci: float = 0.95,
    rng: np.random.Generator | None = None,
) -> tuple[float, float, float]:
    """Bootstrap CI on mean(scores_a - scores_b).

    Parameters
    ----------
    scores_a, scores_b : array-like, shape (n,)
        Paired scores — same seed/layout order.
    reps : int
        Bootstrap resamples (5000 recommended).
    ci : float
        Confidence level, default 0.95.
    rng : np.random.Generator or None
        For reproducibility; uses default_rng(0) if None.

    Returns
    -------
    point : float   mean(d) where d = scores_a - scores_b
    ci_lo : float   lower percentile of bootstrap distribution
    ci_hi : float   upper percentile of bootstrap distribution
    """
    a = np.asarray(scores_a, dtype=float)
    b = np.asarray(scores_b, dtype=float)
    if a.shape != b.shape:
        raise ValueError(f"scores_a and scores_b must match shape; got {a.shape} vs {b.shape}")
    d = a - b
    point = float(np.mean(d))
    rng_ = rng if rng is not None else np.random.default_rng(0)
    n = len(d)
    boot = np.array([np.mean(d[rng_.integers(0, n, size=n)]) for _ in range(reps)])
    alpha = 1.0 - ci
    return point, float(np.percentile(boot, 100 * alpha / 2)), float(np.percentile(boot, 100 * (1 - alpha / 2)))


def prob_of_improvement(
    scores_a: np.ndarray,
    scores_b: np.ndarray,
) -> float:
    """P(a > b) estimated via Mann-Whitney U statistic.

    Follows Agarwal et al. 2021 Section 3: U_a / (n_a * n_b).
    Compatible with 1-D score arrays (n_seeds,).
    """
    from scipy.stats import mannwhitneyu
    a = np.asarray(scores_a, dtype=float).ravel()
    b = np.asarray(scores_b, dtype=float).ravel()
    U, _ = mannwhitneyu(a, b, alternative="greater")
    return float(U / (len(a) * len(b)))
```

```python
def run_baseline(name, cfg_params, seed, n_eval, out, split):
    regime = cfg_params["mobility"]
    env_kw = {k: v for k, v in cfg_params.items() if k not in ("mobility", "label")}
    env = make_env(regime, **env_kw)
    rng = np.random.default_rng(seed)

    if name == "pso":
        policy = TargetTrackingPolicy("pso", seed=seed, replan_every=5)
    elif name == "warm_start_pso":
        policy = TargetTrackingPolicy("warm_start_pso", seed=seed, replan_every=5)
    elif name == "mpc":
        policy = TargetTrackingPolicy("mpc", seed=seed, replan_every=1)
    else:
        raise ValueError(name)

    def act(obs, base):
        if env.unwrapped.t == 0:
            policy.reset(base)
        return policy.act(base)

    covs = [run_episode(env, act, s) for s in layout_seeds(split, n_eval)]
    res = summarize(covs)
    res.update({"method": name, "config": cfg_params["label"], "seed": seed,
                "env_config": env.unwrapped.cfg.to_dict()})
    return res


def run_rl(obs_mode, cfg_name, cfg_params, seed, n_eval, out, split, timesteps):
    regime = cfg_params["mobility"]
    env_kw = {k: v for k, v in cfg_params.items() if k not in ("mobility", "label")}
    model_path = Path(out) / cfg_name / "rl" / "models" / f"seed{seed}.zip"

    if not model_path.exists():
        # train if checkpoint not found
        env_fn = lambda: make_env(regime, obs_mode=obs_mode, **env_kw)
        from stable_baselines3.common.env_util import make_vec_env
        train_env = make_vec_env(env_fn, n_envs=4, seed=seed)
        from scripts.train_rl import _policy_cls_kwargs
        ref_env = make_env(regime, obs_mode=obs_mode, **env_kw)
        pol_cls, pol_kw = _policy_cls_kwargs(obs_mode, ref_env)
        ref_env.close()
        model = PPO(pol_cls, train_env, seed=seed, verbose=0,
                    **({"policy_kwargs": pol_kw} if pol_kw else {}))
        model.learn(total_timesteps=timesteps, progress_bar=False)
        model_path.parent.mkdir(parents=True, exist_ok=True)
        model.save(model_path.with_suffix(""))
    else:
        model = PPO.load(model_path)

    eval_env = make_env(regime, obs_mode=obs_mode, **env_kw)

    def act(obs, _):
        a, _ = model.predict(obs, deterministic=True)
        return a

    covs = [run_episode(eval_env, act, s) for s in layout_seeds(split, n_eval)]
    res = summarize(covs)
    res.update({"method": f"rl_{obs_mode}", "config": cfg_params["label"],
                "seed": seed, "env_config": eval_env.unwrapped.cfg.to_dict()})
    return res

    p.add_argument("--obs-mode", default="flat",
                   choices=["flat", "sorted", "cluster_summary", "spatial_grid", "graph"])
    p.add_argument("--timesteps", type=int, default=200_000)
    p.add_argument("--configs", nargs="+", default=list(CONFIGS.keys()))
    args = p.parse_args()

    for cfg_name in args.configs:
        cfg = CONFIGS[cfg_name]
        print(f"\n=== {cfg_name}: {cfg['label']} ===")
        for seed in args.seeds:
            for method in ["pso", "warm_start_pso", "mpc"]:
                res = run_baseline(method, cfg, seed, args.n_eval, args.out, args.split)
                save_result(args.out, cfg_name, method, seed, res)
                print(f"  {method:16s} seed={seed} final_cov={res['final_cov']:.3f}")
            res = run_rl(args.obs_mode, cfg_name, cfg, seed, args.n_eval,

            save_result(args.out, cfg_name, f"rl_{args.obs_mode}", seed, res)
            print(f"  rl_{args.obs_mode:10s} seed={seed} final_cov={res['final_cov']:.3f}")

    main()
```

```python
import argparse

def load_scores(results_dir: Path, config: str, method: str, metric: str) -> np.ndarray:
    folder = results_dir / config / method
    scores = []
    for f in sorted(folder.glob("seed*.json")):
        d = json.loads(f.read_text())
        scores.append(d[f"{metric}_per_layout"])
    if not scores:
        return np.array([])

    p.add_argument("--results", default="results_rq2")
    p.add_argument("--out", default="figures")

            axes.text(c, r, txt, ha="center", va="center", fontsize=8,
                      color="white" if abs(pt) > vmax * 0.6 else "black")

    plt.colorbar(im, ax=axes, label="mean(RL − PSO)")
    plt.tight_layout()
    out_path = out / "rq2_heatmap.pdf"
    plt.savefig(out_path, bbox_inches="tight")
    print(f"Saved {out_path}")


if __name__ == "__main__":
    main()
```

```python
"""Figure 1: RQ1 representation comparison.

Bar chart: final_cov IQM ± 95% CI for all 5 obs_modes, best RL algo.
Secondary panel: training wall-clock per obs_mode (cost).

Usage:
    python scripts/plot_rq1.py --results results --regime static --out figures/
"""
import argparse
import json
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[1]))

import numpy as np
import matplotlib.pyplot as plt


OBS_MODES = ["flat", "sorted", "cluster_summary", "spatial_grid", "graph"]


def _load_per_layout(results_dir: Path, regime: str, obs_mode: str,
                     metric: str) -> tuple[np.ndarray, list]:
    """Load per-layout final_cov across all seeds for one obs_mode."""
    folder = results_dir / regime
    scores, times = [], []
    for algo_dir in sorted(folder.iterdir()):
        for f in sorted(algo_dir.glob("seed*.json")):
            d = json.loads(f.read_text())
            if d.get("obs_mode") == obs_mode:
                v = d.get(f"{metric}_per_layout")
                if v:
                    scores.append(v)
                t = d.get("train_time_s")
                if t:
                    times.append(t)
    if not scores:
        return np.array([]), times
    # shape: (n_seeds, n_layouts)
    min_len = min(len(s) for s in scores)
    arr = np.array([s[:min_len] for s in scores])
    return arr, times


def iqm(scores: np.ndarray) -> float:
    """Inter-quartile mean: mean of scores in [25th, 75th] percentile."""
    q25, q75 = np.percentile(scores, 25), np.percentile(scores, 75)
    subset = scores[(scores >= q25) & (scores <= q75)]
    return float(np.mean(subset)) if len(subset) > 0 else float(np.mean(scores))


def bootstrap_iqm_ci(scores: np.ndarray, reps: int = 2000, ci: float = 0.95,
                     rng: np.random.Generator | None = None) -> tuple[float, float]:
    rng_ = rng or np.random.default_rng(0)
    n = len(scores)
    q25_fix, q75_fix = np.percentile(scores, 25), np.percentile(scores, 75)
    boot_iqms = []
    for _ in range(reps):
        s = scores[rng_.integers(0, n, size=n)]
        subset = s[(s >= q25_fix) & (s <= q75_fix)]
        boot_iqms.append(float(np.mean(subset)) if len(subset) > 0 else float(np.mean(s)))
    alpha = 1.0 - ci
    return (float(np.percentile(boot_iqms, 100 * alpha / 2)),
            float(np.percentile(boot_iqms, 100 * (1 - alpha / 2))))


def main():
    p = argparse.ArgumentParser()
    p.add_argument("--results", default="results")
    p.add_argument("--regime", default="static")
    p.add_argument("--metric", default="final_cov")
    p.add_argument("--out", default="figures")
    args = p.parse_args()
    results = Path(args.results); out = Path(args.out); out.mkdir(parents=True, exist_ok=True)

    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 5))
    colors = plt.cm.tab10(np.linspace(0, 1, len(OBS_MODES)))

    iqm_vals, ci_los, ci_his, mean_times = [], [], [], []
    for obs_mode in OBS_MODES:
        arr, times = _load_per_layout(results, args.regime, obs_mode, args.metric)
        if arr.size == 0:
            iqm_vals.append(float("nan")); ci_los.append(0); ci_his.append(0)
            mean_times.append(float("nan"))
            continue
        flat = arr.ravel()
        v = iqm(flat)
        lo, hi = bootstrap_iqm_ci(flat)
        iqm_vals.append(v); ci_los.append(v - lo); ci_his.append(hi - v)
        mean_times.append(float(np.mean(times)) / 60.0 if times else float("nan"))

    x = np.arange(len(OBS_MODES))
    bars = ax1.bar(x, iqm_vals, color=colors,
                   yerr=[ci_los, ci_his], capsize=5, error_kw={"linewidth": 1.5})
    ax1.set_xticks(x); ax1.set_xticklabels(OBS_MODES, rotation=15, ha="right")
    ax1.set_ylabel(f"{args.metric} IQM"); ax1.set_ylim(0, 1)
    ax1.set_title(f"(a) RQ1: Representation comparison — {args.regime}")
    ax1.axhline(y=0, color="k", linewidth=0.5)

    ax2.bar(x, mean_times, color=colors)
    ax2.set_xticks(x); ax2.set_xticklabels(OBS_MODES, rotation=15, ha="right")
    ax2.set_ylabel("Training wall-clock (min)"); ax2.set_title("(b) Training cost per obs_mode")

    plt.tight_layout()
    out_path = out / f"rq1_{args.regime}.pdf"
    plt.savefig(out_path, bbox_inches="tight")
    print(f"Saved {out_path}")


if __name__ == "__main__":
    main()
```

```markdown
### Research Questions

**RQ1 — Representation Sensitivity**
```
